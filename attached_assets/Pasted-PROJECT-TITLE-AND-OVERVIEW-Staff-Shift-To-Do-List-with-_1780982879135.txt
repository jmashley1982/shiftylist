PROJECT TITLE AND OVERVIEW
===========================
Staff Shift To-Do List with Google Sheets Logging
- Internal web app for staff to view daily shift tasks, tick off completion (with time logging), and submit final report to Google Sheets.
- Admin panel to manage employees (name + 4-digit code + assigned shift), manage shift task templates (Open, Mid, Close), schedule one‑off tasks for specific employees on specific dates (next 14 days only), view submitted reports (last 30 days) and covering shift workflow.

TECH STACK (exact versions)
============================
- Runtime: Node.js (20.x)
- Framework: Express 4.18.2
- Template engine: EJS 3.1.9
- Database: SQLite3 (better-sqlite3 9.0.0) – stores employees, shifts, task templates, scheduled one‑off tasks.
- Authentication: session-based (express-session 1.17.3) with 4-digit codes.
- Google Sheets API: googleapis 126.0.0 (service account)
- Date handling: date-fns 2.30.0
- Other: dotenv 16.3.1, body-parser 1.20.2

--------------------------------------
```sql
CREATE TABLE IF NOT EXISTS employees (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  code TEXT UNIQUE NOT NULL,
  shift_name TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS shift_templates (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  shift_name TEXT UNIQUE NOT NULL,
  task_text TEXT NOT NULL,
  display_order INTEGER DEFAULT 0
);

CREATE TABLE IF NOT EXISTS scheduled_tasks (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  employee_id INTEGER NOT NULL,
  target_date TEXT NOT NULL, -- YYYY-MM-DD
  task_text TEXT NOT NULL,
  completed_by_admin BOOLEAN DEFAULT 0,
  FOREIGN KEY(employee_id) REFERENCES employees(id)
);

Step 6: Utility – Google Sheets (utils/sheets.js)
js

const { google } = require('googleapis');

async function appendToSheet(submission) {
  const auth = new google.auth.GoogleAuth({
    credentials: {
      client_email: process.env.GOOGLE_SHEETS_CLIENT_EMAIL,
      private_key: process.env.GOOGLE_SHEETS_PRIVATE_KEY.replace(/\\n/g, '\n'),
    },
    scopes: ['https://www.googleapis.com/auth/spreadsheets'],
  });
  const sheets = google.sheets({ version: 'v4', auth });
  const values = [[
    submission.timestamp,
    submission.userCode,
    submission.employeeName,
    submission.shift,
    submission.date,
    JSON.stringify(submission.tasks)
  ]];
  await sheets.spreadsheets.values.append({
    spreadsheetId: process.env.SPREADSHEET_ID,
    range: 'Submissions!A:F',
    valueInputOption: 'USER_ENTERED',
    resource: { values },
  });
}

async function getSubmissionsLast30Days() {
  // Fetch rows from Sheets, filter last 30 days
  const auth = new google.auth.GoogleAuth({
    credentials: {
      client_email: process.env.GOOGLE_SHEETS_CLIENT_EMAIL,
      private_key: process.env.GOOGLE_SHEETS_PRIVATE_KEY.replace(/\\n/g, '\n'),
    },
    scopes: ['https://www.googleapis.com/auth/spreadsheets.readonly'],
  });
  const sheets = google.sheets({ version: 'v4', auth });
  const response = await sheets.spreadsheets.values.get({
    spreadsheetId: process.env.SPREADSHEET_ID,
    range: 'Submissions!A:F',
  });
  const rows = response.data.values || [];
  const thirtyDaysAgo = new Date();
  thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);
  return rows.slice(1).filter(row => new Date(row[0]) >= thirtyDaysAgo);
}

module.exports = { appendToSheet, getSubmissionsLast30Days };

Step 7: Date helpers (utils/dateHelpers.js)

const { format, addDays, isWithinInterval, startOfDay } = require('date-fns');

function getTodayStr() { return format(new Date(), 'yyyy-MM-dd'); }

function getNext14Days() {
  const days = [];
  for (let i = 0; i <= 14; i++) {
    days.push(format(addDays(new Date(), i), 'yyyy-MM-dd'));
  }
  return days;
}

module.exports = { getTodayStr, getNext14Days };

Step 9: Auth routes (routes/auth.js)
js

const express = require('express');
const router = express.Router();
const db = require('../db/init');
const { getTodayStr } = require('../utils/dateHelpers');

router.get('/login', (req, res) => { res.render('login'); });
router.post('/login', (req, res) => {
  const { code } = req.body;
  const employee = db.prepare('SELECT * FROM employees WHERE code = ?').get(code);
  if (!employee) return res.send('Invalid code');
  req.session.pendingCode = code;
  req.session.pendingEmployee = employee;
  res.redirect('/confirm-name');
});
router.get('/confirm-name', (req, res) => {
  if (!req.session.pendingEmployee) return res.redirect('/login');
  res.render('confirmName', { employee: req.session.pendingEmployee });
});
router.post('/confirm-name', (req, res) => {
  if (req.body.confirm === 'yes') {
    const emp = req.session.pendingEmployee;
    req.session.employeeId = emp.id;
    req.session.employeeName = emp.name;
    req.session.employeeShift = emp.shift_name;
    req.session.employeeCode = emp.code;
    delete req.session.pendingEmployee;
    delete req.session.pendingCode;
    res.redirect('/staff/tasks');
  } else {
    res.redirect('/login');
  }
});
// Admin login uses a simple hardcoded route
router.get('/admin/login', (req, res) => { res.render('admin/login'); });
router.post('/admin/login', (req, res) => {
  if (req.body.adminCode === process.env.ADMIN_CODE) {
    req.session.isAdmin = true;
    res.redirect('/admin/dashboard');
  } else { res.send('Invalid admin code'); }
});
router.get('/admin/logout', (req, res) => { req.session.destroy(); res.redirect('/login'); });
module.exports = router;

Step 10: Staff routes (routes/staff.js)
js

const express = require('express');
const router = express.Router();
const db = require('../db/init');
const { ensureStaffAuth } = require('../middleware/auth');
const { getTodayStr } = require('../utils/dateHelpers');
const { appendToSheet } = require('../utils/sheets');

router.get('/tasks', ensureStaffAuth, (req, res) => {
  const empId = req.session.employeeId;
  const today = getTodayStr();
  const employee = db.prepare('SELECT * FROM employees WHERE id = ?').get(empId);
  
  // Check if employee has any scheduled tasks for today OR if their shift template exists for today
  const scheduled = db.prepare('SELECT task_text FROM scheduled_tasks WHERE employee_id = ? AND target_date = ?').all(empId, today);
  const shiftTasks = db.prepare('SELECT task_text FROM shift_templates WHERE shift_name = ? ORDER BY display_order').all(employee.shift_name);
  
  let allTasks = [...shiftTasks.map(t => ({ text: t.task_text, completed: false, completionTime: null })), ...scheduled.map(t => ({ text: t.task_text, completed: false, completionTime: null }))];
  if (allTasks.length === 0) {
    // No tasks assigned for today → prompt covering shift
    return res.render('noTasks', { employeeName: req.session.employeeName });
  }
  res.render('todolist', { tasks: allTasks, employeeName: req.session.employeeName, shift: employee.shift_name });
});

router.post('/submit', ensureStaffAuth, async (req, res) => {
  const { tasks } = req.body; // tasks is JSON string of [{text, completed, completionTime}]
  const submission = {
    timestamp: new Date().toISOString(),
    userCode: req.session.employeeCode,
    employeeName: req.session.employeeName,
    shift: req.session.employeeShift,
    date: getTodayStr(),
    tasks: JSON.parse(tasks)
  };
  await appendToSheet(submission);
  res.send('Submitted to Google Sheet');
});

// Covering shift endpoint – accepts a target employee id
router.post('/cover', ensureStaffAuth, (req, res) => {
  const { targetEmployeeId } = req.body;
  const targetEmp = db.prepare('SELECT * FROM employees WHERE id = ?').get(targetEmployeeId);
  if (!targetEmp) return res.status(400).send('Invalid employee');
  req.session.employeeShift = targetEmp.shift_name;
  req.session.coveringFor = targetEmp.id;
  res.redirect('/staff/tasks');
});
module.exports = router;

Step 11: Admin routes (routes/admin.js)

const express = require('express');
const router = express.Router();
const db = require('../db/init');
const { ensureAdminAuth } = require('../middleware/auth');
const { getNext14Days, getTodayStr } = require('../utils/dateHelpers');
const { getSubmissionsLast30Days } = require('../utils/sheets');

router.use(ensureAdminAuth);

router.get('/dashboard', (req, res) => { res.render('admin/dashboard'); });
// Employees management
router.get('/employees', (req, res) => {
  const employees = db.prepare('SELECT * FROM employees').all();
  const shifts = db.prepare('SELECT DISTINCT shift_name FROM shift_templates').all();
  res.render('admin/employees', { employees, shifts });
});
router.post('/employees/add', (req, res) => {
  const { name, code, shift_name } = req.body;
  db.prepare('INSERT INTO employees (name, code, shift_name) VALUES (?, ?, ?)').run(name, code, shift_name);
  res.redirect('/admin/employees');
});
router.post('/employees/delete/:id', (req, res) => {
  db.prepare('DELETE FROM employees WHERE id = ?').run(req.params.id);
  res.redirect('/admin/employees');
});
// Shift templates
router.get('/shifts', (req, res) => {
  const shifts = db.prepare('SELECT DISTINCT shift_name FROM shift_templates').all();
  const tasks = db.prepare('SELECT * FROM shift_templates ORDER BY shift_name, display_order').all();
  res.render('admin/shifts', { shifts, tasks });
});
router.post('/shifts/addTask', (req, res) => {
  const { shift_name, task_text, display_order } = req.body;
  db.prepare('INSERT INTO shift_templates (shift_name, task_text, display_order) VALUES (?, ?, ?)').run(shift_name, task_text, display_order);
  res.redirect('/admin/shifts');
});
router.post('/shifts/deleteTask/:id', (req, res) => {
  db.prepare('DELETE FROM shift_templates WHERE id = ?').run(req.params.id);
  res.redirect('/admin/shifts');
});
// Schedule one‑off tasks for employees (next 14 days)
router.get('/schedule', (req, res) => {
  const employees = db.prepare('SELECT id, name, shift_name FROM employees').all();
  const futureDates = getNext14Days();
  const scheduled = db.prepare('SELECT * FROM scheduled_tasks').all();
  res.render('admin/schedule', { employees, futureDates, scheduled, today: getTodayStr() });
});
router.post('/schedule/add', (req, res) => {
  const { employee_id, target_date, task_text } = req.body;
  db.prepare('INSERT INTO scheduled_tasks (employee_id, target_date, task_text) VALUES (?, ?, ?)').run(employee_id, target_date, task_text);
  res.redirect('/admin/schedule');
});
router.post('/schedule/delete/:id', (req, res) => {
  db.prepare('DELETE FROM scheduled_tasks WHERE id = ?').run(req.params.id);
  res.redirect('/admin/schedule');
});
// Reports – last 30 days from Google Sheets
router.get('/reports', async (req, res) => {
  const submissions = await getSubmissionsLast30Days();
  res.render('admin/reports', { submissions });
});
module.exports = router;

ADDITIONAL NOTES
================

    Shifts are predefined as "Open (9:30am - 5:00PM)", "Mid (12:00pm - 7:30pm)", "Close (5:00pm - 12:30am)" – store only the short names in DB: "Open", "Mid", "Close".

    Admin can add any custom tasks to any employee for any date within next 14 days via /admin/schedule.

    Covering shift logic: when employee has no tasks for today, they pick a target employee from list, then system loads that employee's shift tasks + their scheduled tasks.

    All submissions go to Google Sheets; admin reports fetch last 30 days from Sheets.

    No automatic deletion – manually remove old rows from Sheets if needed.

    The app includes simple CSS; can be extended.
